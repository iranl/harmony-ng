        Home Assistant Send Command Script
          (scripts.yaml)
        
Copy and paste this script into your Home Assistant
          scripts.yaml to trigger any IR or Bluetooth command with queued concurrency, delays, and
          hold/pulse time:
        

        
harmony_send_command:
  alias: "Harmony Hub Send Command"
  description: "Send IR/BT command to device via Harmony Hub MQTT"
  mode: queued
  max: 20
  fields:
    device:
      description: "Target device name or ID"
      example: "Tv Samsung"
      required: true
      selector:
        text: {}
    command:
      description: "Command or list of commands"
      example: "InputHdmi2"
      required: true
    delay_secs:
      description: "Delay between commands in seconds"
      example: 0.4
      default: 0.4
      selector:
        number:
          min: 0
          max: 10
          step: 0.1
          unit_of_measurement: "s"
    hold_secs:
      description: "Hold duration in seconds"
      example: 0
      default: 0
      selector:
        number:
          min: 0
          max: 5
          step: 0.1
          unit_of_measurement: "s"
    num_repeats:
      description: "Times to repeat command"
      example: 1
      default: 1
      selector:
        number:
          min: 1
          max: 20
          step: 1
  sequence:
    - action: mqtt.publish
      data:
        topic: "harmony/hub/send_command"
        payload: >-
          {{ {
            'device': device,
            'command': [command] if command is string else command,
            'delay_secs': delay_secs | default(0.4, true) | float,
            'hold_secs': hold_secs | default(0, true) | float,
            'num_repeats': num_repeats | default(1, true) | int
          } | to_json }}
        

        

          Example HA automation action: action: script.harmony_send_command with
          device: "Samsung TV" and command: ["PowerToggle"].
